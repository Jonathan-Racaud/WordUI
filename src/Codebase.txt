───── ./Core/Layout.bf ─────
     1	using System;
     2	
     3	namespace WordUI;
     4	
     5	// Flex layout pass: constraints-down, sizes-up, sum-main/max-cross,
     6	// leftover→Spacer. Runs after Build(), stores a Rect on every widget.
     7	static class Layout
     8	{
     9		// Entry point: lay out the entire tree within the given rect.
    10		public static void Run(Rect available, Widget root)
    11		{
    12			LayoutWidget(available, root);
    13		}
    14	
    15		static void LayoutWidget(Rect available, Widget widget)
    16		{
    17			widget.mRect = available;
    18	
    19			let children = widget.GetComponent<Children>();
    20			if (children == null)
    21				return;
    22	
    23			let stack = widget.GetComponent<StackLayout>();
    24			if (stack == null)
    25				return;
    26	
    27			// Compute inner rect (available minus padding)
    28			float pad = widget.Padding;
    29			Rect inner = .(
    30				available.X + pad,
    31				available.Y + pad,
    32				available.Width - 2 * pad,
    33				available.Height - 2 * pad
    34			);
    35	
    36			bool isVertical = stack.Axis == .Vertical;
    37	
    38			// Count spacers and measure non-spacer children
    39			int spacerCount = 0;
    40			int itemCount = children.Items.Count;
    41			float[] mainSizes = scope float[itemCount];
    42			float[] crossSizes = scope float[itemCount];
    43	
    44			float totalMain = 0;
    45			float maxCross = 0;
    46	
    47			for (int i = 0; i < itemCount; i++)
    48			{
    49				let child = children.Items[i];
    50				if (child.Name == "Spacer")
    51				{
    52					spacerCount++;
    53					mainSizes[i] = -1; // marker
    54					continue;
    55				}
    56	
    57				Size constraint = isVertical
    58					? .(inner.Width, float.MaxValue)
    59					: .(float.MaxValue, inner.Height);
    60	
    61				let size = MeasureWidget(child, constraint);
    62				float main = isVertical ? size.Height : size.Width;
    63				float cross = isVertical ? size.Width : size.Height;
    64				mainSizes[i] = main;
    65				crossSizes[i] = cross;
    66				totalMain += main;
    67				maxCross = Math.Max(maxCross, cross);
    68			}
    69	
    70			// Spacing between non-spacer items
    71			int nonSpacerCount = itemCount - spacerCount;
    72			if (nonSpacerCount > 1)
    73				totalMain += stack.Spacing * (nonSpacerCount - 1);
    74	
    75			// Distribute leftover main-axis space to spacers
    76			float mainAvailable = isVertical ? inner.Height : inner.Width;
    77			float leftover = mainAvailable - totalMain;
    78			float spacerSize = spacerCount > 0
    79				? Math.Max(0, leftover / spacerCount)
    80				: 0;
    81	
    82			// Assign rects to children
    83			float mainPos = isVertical ? inner.Y : inner.X;
    84	
    85			for (int i = 0; i < itemCount; i++)
    86			{
    87				let child = children.Items[i];
    88	
    89				if (child.Name == "Spacer")
    90				{
    91					child.mRect = isVertical
    92						? Rect(inner.X, mainPos, inner.Width, spacerSize)
    93						: Rect(mainPos, inner.Y, spacerSize, inner.Height);
    94					mainPos += spacerSize;
    95					continue;
    96				}
    97	
    98				float childMain = mainSizes[i];
    99				float childCrossAvailable = isVertical ? inner.Width : inner.Height;
   100				float childCrossSize = crossSizes[i];
   101	
   102				// Apply cross-axis alignment from LayoutHint (default: Stretch).
   103				// A sizing constraint on the cross axis overrides Stretch so that
   104				// e.g. Height(.Fixed(20)) inside an HStack produces a 20px-tall
   105				// child instead of forcing it to fill the row height.
   106				float childCrossPos;
   107				float childCrossUsed;
   108				var hint = child.GetProperty<LayoutHint>();
   109				bool hasCrossConstraint = hint != null &&
   110					(isVertical ? !(hint.Width case .Unconstrained) : !(hint.Height case .Unconstrained));
   111	
   112				if (hint != null && hint.CrossAlign != .Stretch)
   113				{
   114					childCrossUsed = childCrossSize;
   115					if (hint.CrossAlign == .Center)
   116						childCrossPos = (isVertical ? inner.X : inner.Y) + (childCrossAvailable - childCrossSize) / 2;
   117					else if (hint.CrossAlign == .End)
   118						childCrossPos = (isVertical ? inner.X : inner.Y) + childCrossAvailable - childCrossSize;
   119					else // .Start
   120						childCrossPos = isVertical ? inner.X : inner.Y;
   121				}
   122				else if (hasCrossConstraint)
   123				{
   124					// Sizing constraint overrides Stretch; position at start
   125					childCrossUsed = childCrossSize;
   126					childCrossPos = isVertical ? inner.X : inner.Y;
   127				}
   128				else
   129				{
   130					childCrossPos = isVertical ? inner.X : inner.Y;
   131					childCrossUsed = childCrossAvailable;
   132				}
   133	
   134				Rect childRect = isVertical
   135					? Rect(childCrossPos, mainPos, childCrossUsed, childMain)
   136					: Rect(mainPos, childCrossPos, childMain, childCrossUsed);
   137	
   138				LayoutWidget(childRect, child); // recurse
   139	
   140				mainPos += childMain + stack.Spacing;
   141			}
   142		}
   143	
   144		// Measure a widget's intrinsic size. For containers with StackLayout,
   145		// measures children and computes the total; for leaves, asks components.
   146		static Size MeasureWidget(Widget widget, Size constraint)
   147		{
   148			let children = widget.GetComponent<Children>();
   149			let stack = widget.GetComponent<StackLayout>();
   150	
   151			// Container: measure children and compute flex total
   152			if (children != null && stack != null)
   153			{
   154				float pad = widget.Padding;
   155				bool isVertical = stack.Axis == .Vertical;
   156	
   157				float totalMain = 0;
   158				float maxCross = 0;
   159				int spacerCount = 0;
   160	
   161				for (let child in children.Items)
   162				{
   163					if (child.Name == "Spacer")
   164					{
   165						spacerCount++;
   166						continue;
   167					}
   168	
   169					Size childConstraint = isVertical
   170						? .(constraint.Width - 2 * pad, float.MaxValue)
   171						: .(float.MaxValue, constraint.Height - 2 * pad);
   172	
   173					let size = MeasureWidget(child, childConstraint);
   174					totalMain += isVertical ? size.Height : size.Width;
   175					maxCross = Math.Max(maxCross, isVertical ? size.Width : size.Height);
   176				}
   177	
   178				// Spacing between non-spacer items
   179				int nonSpacerCount = children.Items.Count - spacerCount;
   180				if (nonSpacerCount > 1)
   181					totalMain += stack.Spacing * (nonSpacerCount - 1);
   182	
   183				float w = isVertical ? maxCross : totalMain;
   184				float h = isVertical ? totalMain : maxCross;
   185				w += pad * 2;
   186				h += pad * 2;
   187				var hint = widget.GetProperty<LayoutHint>();
   188				if (hint != null)
   189				{
   190					w = hint.Width.Apply(w, constraint.Width);
   191					h = hint.Height.Apply(h, constraint.Height);
   192				}
   193				return .(w, h);
   194			}
   195	
   196			// Leaf: ask components
   197			float cw = 0, ch = 0;
   198			for (let c in widget.Components)
   199			{
   200				let size = c.Measure(constraint);
   201				cw = Math.Max(cw, size.Width);
   202				ch = Math.Max(ch, size.Height);
   203			}
   204			cw += widget.Padding * 2;
   205			ch += widget.Padding * 2;
   206			var hint = widget.GetProperty<LayoutHint>();
   207			if (hint != null)
   208			{
   209				cw = hint.Width.Apply(cw, constraint.Width);
   210				ch = hint.Height.Apply(ch, constraint.Height);
   211			}
   212			return .(cw, ch);
   213		}
   214	}

     1	using System;
     2	
     3	namespace WordUI;

     5	// Shared hit-test utility: given a point in window coordinates, walks the
     6	// widget tree and returns the leaf-most widget whose mRect contains the
     7	// point. Returns null if no widget is hit.
     8	static class HitTest
     9	{
    10		public static Widget At(Widget root, float x, float y)
    11		{
    12			return Find(root, x, y, null);
    13		}
    14	
    15		static Widget Find(Widget w, float x, float y, Widget best)
    16		{
    17			if (w == null)
    18				return best;
    19	
    20			Widget current = best;
    21			if (x >= w.mRect.X && x < w.mRect.Right &&
    22				y >= w.mRect.Y && y < w.mRect.Bottom)
    23			{
    24				current = w;
    25			}
    26	
    27			let children = w.GetComponent<Children>();
    28			if (children != null)
    29			{
    30				for (let child in children.Items)
    31				{
    32					current = Find(child, x, y, current);
    33				}
    34			}
    35			return current;
    36		}
    37	}

     1	using System;
     2	using System.Collections;
     3	
     4	namespace WordUI;
     5	
     6	// Rebuild-on-invalidate driver: `Build()` re-runs only after `Invalidate()`,
     7	// producing a fresh widget tree in a bump arena; the previous tree (widgets,
     8	// components, string copies) is freed wholesale by deleting the arena.
     9	//
    10	// The retained Node tree lives on the heap and survives rebuilds. After each
    11	// build, ReconcileTree() matches Widget descriptions (by StableKey) against
    12	// existing Nodes: matched Nodes are reused and their state (hover, press,
    13	// scroll, animations) persists; new Nodes are created; orphaned Nodes are
    14	// disposed.
    15	//
    16	// Non-core conveniences (e.g. DumpTree) live in Diagnostics/ as extensions
    17	// of this class.
    18	public abstract class App
    19	{
    20		// The active App instance; set by SDLApp constructor, used by trackers
    21		// and other infrastructure that need to trigger rebuilds.
    22		public static App Instance;
    23	
    24		BumpAllocator mArena ~ delete _;
    25		// Owns event handlers (click, drag, …) the user allocates with `new`; freed
    26		// wholesale on the next rebuild. Delegate is the common base of every handler
    27		// signature (ClickHandler, DragHandler).
    28		List<Delegate> mHandlers = new .() ~ { ClearHandlers(); delete _; };
    29		Widget mRoot;
    30		bool mDirty = true;
    31		bool mNeedsRepaint = false;
    32		bool mNeedsLayout = false;
    33	
    34		// ---- Retained runtime: the Node tree ----
    35	
    36		// Root of the retained node tree. Nodes are heap-allocated and survive
    37		// rebuilds; they carry identity, parent/child relationships, generic state
    38		// storage, and per-node animation clocks.
    39		Node mRootNode;
    40	
    41		// O(1) StableKey → Node lookup so trackers and draw code can access
    42		// persistent state without a tree walk.
    43		Dictionary<int, Node> mNodeIndex = new .() ~ delete _;
    44	
    45		// ---- Drag / pointer capture (per-pointer-session) ----
    46	
    47		// The active capture state. InputDispatcher reads this via CaptureActive/CaptureKey;
    48		// writes via CaptureBegin/CaptureUpdate/CaptureEnd so the struct field (not a copy)
    49		// is mutated.
    50		PointerCapture mCapture;
    51	
    52		// ---- Focus ----
    53	
    54		// StableKey of the currently focused widget, or 0 if none.
    55		// Set by FocusHandler on MouseDown; read by InputDispatcher for key routing
    56		// and by KeyboardHandler to check if its widget has focus.
    57		public int FocusedKey;
    58	
    59		// Returns true while any IInputHandler holds a pointer capture.
    60		public bool CaptureActive => mCapture.Active;
    61	
    62		// Returns the StableKey of the capturing widget, or 0.
    63		public int CaptureKey => mCapture.Key;
    64	
    65		// Forwarding mutators that directly mutate mCapture fields (not struct copies).
    66		public void CaptureBegin(int key, float x, float y)
    67		{
    68			mCapture.Key = key;
    69			mCapture.LastX = x;
    70			mCapture.LastY = y;
    71		}
    72		public (float dx, float dy) CaptureUpdate(float x, float y)
    73		{
    74			let dx = x - mCapture.LastX;
    75			let dy = y - mCapture.LastY;
    76			mCapture.LastX = x;
    77			mCapture.LastY = y;
    78			return (dx, dy);
    79		}
    80		public void CaptureEnd()
    81		{
    82			mCapture.Key = 0;
    83		}
    84	
    85		// ---- Event trackers (registered by the backend) ----
    86	
    87		// Trackers are called in registration order for each input event.
    88		// A tracker returning true stops iteration (consumes the event).
    89		List<ITracker> mTrackers = new .() ~ DeleteContainerAndItems!(_);
    90	
    91		public void RegisterTracker(ITracker t)
    92		{
    93			mTrackers.Add(t);
    94		}
    95	
    96		// Iterated by the backend's event loop. Registration order = dispatch order.
    97		public List<ITracker> Trackers => mTrackers;
    98	
    99		// ---- Public accessors ----
   100	
   101		protected Widget Root => mRoot;
   102	
   103		protected abstract void Build();
   104	
   105		// App-data changed: rebuild + layout + render on the next frame.
   106		public void Invalidate() { mDirty = true; }
   107	
   108		// Visual-only change (interaction state, animation tick): render without a
   109		// rebuild. Cheap path — no arena teardown, no layout. Also marks the root
   110		// node paint-dirty so per-node dirty tracking stays in sync.
   111		public void RequestRepaint()
   112		{
   113			mNeedsRepaint = true;
   114			if (mRootNode != null)
   115				mRootNode.MarkNeedsPaint();
   116		}
   117	
   118		// Geometry-only change (resize, content reflow): re-run layout without
   119		// tearing down the arena or rebuilding the widget tree. Cheaper than a
   120		// full rebuild when structure hasn't changed.
   121		public void RequestLayout()
   122		{
   123			mNeedsLayout = true;
   124			if (mRootNode != null)
   125				mRootNode.MarkNeedsLayout();
   126		}
   127	
   128		public bool NeedsRepaint => mNeedsRepaint;
   129		public void ClearRepaint() { mNeedsRepaint = false; }
   130	
   131		// Runs Layout on the existing widget tree if mNeedsLayout is set.
   132		// Returns true when layout was performed (so the caller knows to re-render).
   133		public bool LayoutIfNeeded(Rect available)
   134		{
   135			if (!mNeedsLayout || mRoot == null)
   136				return false;
   137			Layout.Run(available, mRoot);
   138			if (mRootNode != null)
   139				mRootNode.ClearDirty();
   140			mNeedsLayout = false;
   141			return true;
   142		}
   143	
   144		// ---- Node lookup ----
   145	
   146		// Returns the retained Node for a given StableKey, or null if none exists.
   147		// O(1) via the index. Used by trackers and draw code to read/write
   148		// persistent per-widget state.
   149		public Node FindNode(int stableKey)
   150		{
   151			Node node;
   152			mNodeIndex.TryGetValue(stableKey, out node);
   153			return node;
   154		}
   155	
   156		// ---- Animation ticks ----
   157	
   158		// Wall-clock base for computing per-frame deltas. Initialised to -1 so the
   159		// first TickAll computes dtMs=0 (no jump). dtMs is computed once per frame
   160		// and passed uniformly to the whole retained tree.
   161		int mLastTickMs = -1;
   162	
   163		// Advances every ITickable in the retained Node tree by the wall-clock
   164		// delta since the last tick. Called once per frame from the event loop.
   165		public void TickAll(int nowMs)
   166		{
   167			if (mLastTickMs < 0)
   168				mLastTickMs = nowMs;
   169			int dtMs = nowMs - mLastTickMs;
   170			mLastTickMs = nowMs;
   171			if (dtMs <= 0 || mRootNode == null)
   172				return;

   184					return true;

   186			}
   187		}

   189		// ---- Build lifecycle ----
   190	
   191		// Returns true if a rebuild happened.
   192		public bool RebuildIfNeeded()
   193		{
   194			if (!mDirty)
   195				return false;
   196	
   197			mRoot = null;
   198			DeleteAndNullify!(mArena);   // frees the entire previous tree
   199			ClearHandlers();
   200			mArena = new BumpAllocator();
   201	
   202			UI.BeginBuild(mArena, mHandlers);
   203			Build();
   204			mRoot = UI.EndBuild();
   205	
   206			// Assign stable identity, collecting the live key set.
   207			HashSet<int> live = scope .();
   208			KeyAssign.Run(mRoot, live);
   209	
   210			// Reconcile the retained Node tree against the new Widget tree:
   211			// reuse matched Nodes, create new ones, dispose orphans.
   212			ReconcileTree(mRoot, live);
   213	
   214			mDirty = false;
   215			mNeedsLayout = false;  // rebuild implies relayout; clear stale flag
   216			return true;
   217		}
   218	
   219		// ---- Reconciliation ----
   220	
   221		// Matches the freshly-built Widget tree against the existing retained Node
   222		// tree by StableKey. Matched Nodes are reused (their persistent state
   223		// survives); new Nodes are created and indexed; orphaned Nodes (in the index
   224		// but not in the live set) are disposed and removed.
   225		void ReconcileTree(Widget root, HashSet<int> live)
   226		{
   227			// Walk the Widget tree depth-first, building or reusing Nodes.
   228			Node newRoot = ReconcileWidget(root, null, live);
   229	
   230			// Dispose orphans: any Node in the index that wasn't seen this build.
   231			List<int> orphanKeys = scope .();
   232			for (let pair in mNodeIndex)
   233			{
   234				if (!live.Contains(pair.key))
   235					orphanKeys.Add(pair.key);
   236			}
   237			for (let key in orphanKeys)
   238			{
   239				let node = mNodeIndex[key];
   240				mNodeIndex.Remove(key);
   241				node.Dispose();
   242				delete node;
   243			}
   244	
   245			mRootNode = newRoot;
   246		}
   247	
   248		// Recursively reconciles one Widget to a retained Node. Returns the
   249		// (possibly new, possibly reused) Node for this subtree.
   250		Node ReconcileWidget(Widget w, Node parent, HashSet<int> live)
   251		{
   252			if (w == null)
   253				return null;
   254	
   255			bool isNew = false;
   256			Node node;
   257			if (!mNodeIndex.TryGetValue(w.StableKey, out node))
   258			{
   259				// First time seeing this key: create a fresh Node.
   260				node = new Node(w.StableKey);
   261				mNodeIndex[w.StableKey] = node;
   262				isNew = true;
   263			}
   264	
   265			// Update parent/child links (the Widget tree structure may have changed).
   266			node.Parent = parent;
   267			node.Children.Clear();
   268	
   269			// Collect live slot ids from IInteractionState properties (for animation
   270			// clock keys) and IHasStateSlots behaviors (for state slot keys on the
   271			// retained Node). These drive pruning and lifecycle dispatch — only slots
   272			// declared by actual capabilities on this widget survive pruning.
   273			HashSet<int> liveSlots = scope .();
   274	
   275			// The hover slot is managed globally by InputDispatcher (not a per-widget
   276			// behavior), so it's always preserved regardless of widget capabilities.
   277			liveSlots.Add(StateSlots.Hover.Id);
   278	
   279			for (let p in w.Properties)
   280			{
   281				let s = p as IInteractionState;
   282				if (s != null)
   283					liveSlots.Add(s.StateId);
   284			}
   285			for (let b in w.Behaviors)
   286			{
   287				let hs = b as IHasStateSlots;
   288				if (hs != null)
   289					hs.CollectSlots(liveSlots);
   290			}
   291	
   292			// Prune stale state before lifecycle dispatch: OnCleanup fires inside
   293			// PruneState for every slot being removed, so the capability can read
   294			// its data one last time.
   295			node.PruneState(liveSlots);
   296	
   297			// Fire lifecycle callbacks (OnCreate / OnReuse) top-down, before children
   298			// are reconciled, so a parent capability can initialise before its subtree.
   299			node.OnReconciled(isNew, liveSlots);
   300	
   301			// Recurse into children.
   302			let children = w.GetComponent<Children>();
   303			if (children != null)
   304			{
   305				for (let child in children.Items)
   306				{
   307					let childNode = ReconcileWidget(child, node, live);
   308					if (childNode != null)
   309						node.Children.Add(childNode);
   310				}
   311			}
   312	
   313			return node;
   314		}
   315	
   316		void ClearHandlers()
   317		{
   318			for (let h in mHandlers)
   319				delete h;
   320			mHandlers.Clear();
   321		}

     5	// Time-warp curves for animation. An easing maps normalized progress p in
     6	// [0,1] (linear time) to eased progress, also in [0,1]. The resolver feeds the
     7	// eased value to Lerp. Curves live here so both Transition words and the
     8	// resolver share one definition; user code can add cases via the enum + Apply.
     9	public enum Easing
    10	{
    11		case Linear;
    12		case EaseInOut;
    13		case Spring;
    14	
    15		// Maps linear progress p (already clamped to [0,1]) to eased progress.
    16		public float Apply(float p)
    17		{
    18			switch (this)
    19			{
    20			case .Linear:
    21				return p;
    22			case .EaseInOut:
    23				// smoothstep: 3p^2 - 2p^3
    24				return p * p * (3f - 2f * p);
    25			case .Spring:
    26				// Lightly overshooting settle; still lands on 1 at p==1.
    27				let s = 1.70158f;
    28				let x = p - 1f;
    29				return 1f + (s + 1f) * x * x * x + s * x * x;
    30			}
    31		}
    32	}
    33	
    34	// Scalar linear interpolation shared by the resolver for float channels
    35	// (FontSize, StrokeWidth, Radius, drag offsets).
    36	static class Lerp
    37	{
    38		public static float F(float a, float b, float t) => a + (b - a) * t;
    39	}

     6	// Assigns Widget.StableKey by a deterministic walk of the finished tree, so the
     7	// same logical widget gets the same key across rebuilds (as long as the build
     8	// structure is stable). This is the identity persistent interaction state keys
     9	// off of — it replaces the old dangling Widget* / positional build ordinal.
    10	//
    11	// Each widget's key is a path hash: hash(parentKey, Name, discriminator), where
    12	// the discriminator is the author's Key() hint when set (>= 0, e.g. list item
    13	// index) or the sibling ordinal otherwise. Feeding the explicit hint into the
    14	// hash (rather than using it raw) keeps user keys and auto keys in one space
    15	// with no collisions between a small user int and a computed path.
    16	static class KeyAssign
    17	{
    18		// When `liveKeys` is provided, every assigned StableKey is added to it, so the
    19		// caller can evict store entries for keys absent from this build.
    20		public static void Run(Widget root, HashSet<int> liveKeys = null)
    21		{
    22			if (root != null)
    23				Assign(root, 1, 0, liveKeys);
    24		}
    25	
    26		static void Assign(Widget w, int parentKey, int siblingIndex, HashSet<int> live)
    27		{
    28			int disc = w.Key >= 0 ? w.Key : siblingIndex;
    29			w.StableKey = Mix(Mix(parentKey, HashName(w.Name)), disc);
    30			if (live != null)
    31				live.Add(w.StableKey);
    32	
    33			let children = w.GetComponent<Children>();
    34			if (children == null)
    35				return;
    36			for (int i = 0; i < children.Items.Count; i++)
    37				Assign(children.Items[i], w.StableKey, i, live);
    38		}
    39	
    40		// FNV-1a over the name bytes; stable and allocation-free.
    41		static int HashName(StringView name)
    42		{
    43			int h = (int)0xcbf29ce484222325UL;
    44			for (let c in name)
    45				h = (h ^ (int)c) &* (int)0x100000001b3UL;
    46			return h;
    47		}
    48	
    49		static int Mix(int a, int b)
    50		{
    51			int h = a &* (int)0x100000001b3UL;
    52			return h ^ b;
    53		}
    54	}

    77		// (ox,oy) is the accumulated render-time offset from ancestor offsets
    78		// (drag, scroll, etc.). Each component on a widget may contribute an offset
    79		// via GetRenderOffset — the draw loop accumulates all of them into the
    80		// child rendering origin. Zero hardcoded knowledge of specific interaction
    81		// types in the draw pipeline.
    82		void DrawWidget(Widget w, float ox, float oy)
    83		{
    84			if (w == null)
    85				return;
    86	
    87			let app = App.Instance;
    88			let node = app.FindNode(w.StableKey);
    89	
    90			// Accumulate render offsets from every component on this widget.
    91			var tx = ox;
    92			var ty = oy;
    93			for (let c in w.Components)
    94			{
    95				let (cox, coy) = c.GetRenderOffset(w, node);
    96				tx += cox;
    97				ty += coy;
    98			}
    99	
   100			Rect bounds = .(w.mRect.X + tx, w.mRect.Y + ty, w.mRect.Width, w.mRect.Height);
   101	
   102			// 1. Background — fills behind everything (Interior)
   103			for (let c in w.Components)
   104				c.DrawBackground(mPainter, w, bounds);
   105	
   106			// 2. Content — widget's own content (TextStyle, custom drawing)
   107			for (let c in w.Components)
   108				c.DrawContent(mPainter, w, bounds);
   109	
   110			// 3. Children — recurse, optionally clipped to this widget's box.
   111			var childOx = tx, childOy = ty;
   112	
   113			bool clip = w.GetComponent<ClipRegion>() != null;
   114			if (clip)
   115				mPainter.PushClip(bounds);
   116	
   117			let children = w.GetComponent<Children>();
   118			if (children != null)
   119			{
   120				for (let child in children.Items)
   121					DrawWidget(child, childOx, childOy);
   122			}
   123	
   124			if (clip)
   125				mPainter.PopClip();
   126	
   127			// 4. Overlay — strokes on top of content and children (Boundary),
   128			// unclipped so a container's own border draws around its full box.
   129			for (let c in w.Components)
   130				c.DrawOverlay(mPainter, w, bounds);
   131		}

   133		// Enter the event loop. Returns when the window is closed.
   134		//
   135		// Frame-batched and event-driven: when nothing is animating or dragging the
   136		// loop BLOCKS in SDL_WaitEvent (true zero idle CPU); while a transition or
   137		// drag is in flight it uses a ~16ms timeout as a frame clock. After each wake
   138		// it drains all pending events (coalescing motion), advances animation
   139		// clocks, then rebuilds+lays-out only on mDirty, else repaints only.
   140		public void Run()
   141		{
   142			RebuildIfNeeded();
   143			Layout.Run(Rect(0, 0, mWidth, mHeight), Root);
   144			DumpTree();
   145			Render();
   146	
   147			bool quit = false;
   148			SDL_Event event = .();
   149			while (!quit)
   150			{
   151				// Block when idle; tick at ~60fps while animating/dragging.
   152				if (App.Instance.NeedsTick)
   153				{
   154					if (SDL_WaitEventTimeout(&event, 16))
   155						HandleEvent(event, ref quit);
   156				}
   157				else if (SDL_WaitEvent(&event))
   158				{
   159					HandleEvent(event, ref quit);
   160				}
   161	
   162				// Drain the rest of the queue, coalescing motion to the last position.
   163				bool hadMotion = false;
   164				float motionX = 0, motionY = 0;
   165				while (SDL_PollEvent(&event))
   166				{
   167					if (event.type == (uint32)SDL_EventType.SDL_EVENT_MOUSE_MOTION)
   168					{
   169						hadMotion = true;
   170						motionX = event.motion.x; motionY = event.motion.y;
   171						continue;
   172					}
   173					HandleEvent(event, ref quit);
   174				}
   175				if (hadMotion)
   176					HandleMotion(motionX, motionY);
   177	
   178				// Advance animation clocks against wall-clock time.
   179				App.Instance.TickAll((int)SDL_GetTicks());
   180	
   181				if (RebuildIfNeeded())
   182				{
   183					Layout.Run(Rect(0, 0, mWidth, mHeight), Root);
   184					Render();
   185				}
   186				else if (LayoutIfNeeded(Rect(0, 0, mWidth, mHeight)))
   187				{
   188					Render();
   189				}
   190				else if (NeedsRepaint || App.Instance.NeedsTick)
   191				{
   192					Render();
   193					ClearRepaint();
   194				}
   195			}
   196		}
   197	
   198		void HandleEvent(SDL_Event event, ref bool quit)
   199		{
   200			switch (event.type)
   201			{
   202			case (uint32)SDL_EventType.SDL_EVENT_QUIT:
   203				quit = true;
   204			case (uint32)SDL_EventType.SDL_EVENT_WINDOW_RESIZED:
   205				mWidth = (int32)event.window.data1;
   206				mHeight = (int32)event.window.data2;
   207				RequestLayout();  // geometry changed, structure didn't — no rebuild needed
   208			case (uint32)SDL_EventType.SDL_EVENT_MOUSE_MOTION:
   209				HandleMotion(event.motion.x, event.motion.y);
   210			case (uint32)SDL_EventType.SDL_EVENT_MOUSE_WHEEL:
   211				DispatchEvent(.() {
   212					Type = .MouseWheel,
   213					X = event.wheel.mouse_x, Y = event.wheel.mouse_y,
   214					WheelX = event.wheel.x, WheelY = event.wheel.y
   215				});
   216			case (uint32)SDL_EventType.SDL_EVENT_MOUSE_BUTTON_DOWN:
   217				if (event.button.button == (uint8)SDL_MouseButtonFlags.SDL_BUTTON_LEFT)
   218					DispatchEvent(.() { Type = .MouseDown, X = event.button.x, Y = event.button.y });
   219			case (uint32)SDL_EventType.SDL_EVENT_MOUSE_BUTTON_UP:
   220				if (event.button.button == (uint8)SDL_MouseButtonFlags.SDL_BUTTON_LEFT)
   221					DispatchEvent(.() { Type = .MouseUp, X = event.button.x, Y = event.button.y });
   222			case (uint32)SDL_EventType.SDL_EVENT_KEY_DOWN:
   223				DispatchEvent(.() {
   224					Type = .KeyDown,
   225					KeyCode = (int)event.key.key,
   226					KeyModifiers = (int)event.key.mod
   227				});
   228			default:
   229			}
   230		}

   232		// Dispatch an event through registered trackers in priority order.
   233		// A tracker returning true stops iteration (consumes the event).
   234		void DispatchEvent(InputEvent input)
   235		{
   236			for (let t in App.Instance.Trackers)

   245			for (let t in App.Instance.Trackers)

   247					return;

     5	// Draw-time style resolution. Drawing components call these with their base
     6	// value; the resolver blends every active (or still-animating) patch over it,
     7	// field by field, so concurrent states compose. Animated fields interpolate via
     8	// the persistent clock in the retained Node tree, which is what lets a transition
     9	// survive a rebuild and keep the frame loop ticking.
    10	enum ColorField { Fill, Stroke, TextColor }
    11	enum FloatField { FontSize, StrokeWidth }
    12	
    13	static class StyleResolve
    14	{
    15		public static Color Color(Widget w, Color baseValue, ColorField field)
    16		{
    17			let style = w.GetProperty<InteractionStyle>();
    18			if (style == null || style.Records == null)
    19				return baseValue;
    20	
    21			Color result = baseValue;
    22			for (let rec in style.Records)
    23			{
    24				if (!HasColor(rec.Patch, field, let target, let anim))
    25					continue;
    26				let factor = Factor(w, rec, anim);
    27				if (factor <= 0f)
    28					continue;
    29				result = WordUI.Color.Lerp(result, target, factor);
    30			}
    31			return result;
    32		}
    33	
    34		public static float Float(Widget w, float baseValue, FloatField field)
    35		{
    36			let style = w.GetProperty<InteractionStyle>();
    37			if (style == null || style.Records == null)
    38				return baseValue;
    39	
    40			float result = baseValue;
    41			for (let rec in style.Records)
    42			{
    43				if (!HasFloat(rec.Patch, field, let target, let anim))
    44					continue;
    45				let factor = Factor(w, rec, anim);
    46				if (factor <= 0f)
    47					continue;
    48				result = Lerp.F(result, target, factor);
    49			}
    50			return result;
    51		}
    52	
    53		// Blend strength for one record: 1/0 instantly for non-animated fields,
    54		// or the eased clock value for animated ones (which also advances/keeps the
    55		// loop alive while settling). Animation clocks live in the retained
    56		// AnimationState tickable on each Node, not in Node itself.
    57		static float Factor(Widget w, PatchRecord rec, bool anim)
    58		{
    59			let active = rec.State.IsActiveFor(w.StableKey);
    60			if (!anim)
    61				return active ? 1f : 0f;
    62			let node = App.Instance.FindNode(w.StableKey);
    63			let clocks = AnimationState.Get(node);
    64			return clocks != null
    65				? clocks.Blend(rec.State.StateId, active, rec.Patch.DurationMs, rec.Patch.Curve)
    66				: 0f;
    67		}
    68	
    69		static bool HasColor(StatePatch p, ColorField field, out Color value, out bool anim)
    70		{
    71			switch (field)
    72			{
    73			case .Fill:      value = p.Fill;      anim = p.FillAnim;      return p.HasFill;
    74			case .Stroke:    value = p.Stroke;    anim = p.StrokeAnim;    return p.HasStroke;
    75			case .TextColor: value = p.TextColor; anim = p.TextColorAnim; return p.HasTextColor;
    76			}
    77		}
    78	
    79		static bool HasFloat(StatePatch p, FloatField field, out float value, out bool anim)
    80		{
    81			switch (field)
    82			{
    83			case .FontSize:    value = p.FontSize;    anim = p.FontSizeAnim;    return p.HasFontSize;
    84			case .StrokeWidth: value = p.StrokeWidth; anim = p.StrokeWidthAnim; return p.HasStrokeWidth;
    85			}
    86		}

     6	// The retained runtime element: a long-lived, heap-allocated node that persists
     7	// across rebuilds. While Widgets are transient descriptions rebuilt every frame,
     8	// Nodes provide stable identity, tree structure, lifecycle, and generic per-node
     9	// state storage that capabilities (hover, press, scroll, animation) write into
    10	// and read from.
    11	//
    12	// Node knows NOTHING about what any state slot means. Slot constants are defined
    13	// by the capabilities that use them (framework reserves 1..99; user capabilities
    14	// pick >= 100). The storage is a single Dictionary<int, Variant> — any type fits,
    15	// small value types (bool, int, float) are stored inline with zero allocation.
    16	//
    17	// Tick dispatch is generic: capabilities that need per-frame time updates implement
    18	// ITickable and register themselves via RegisterTickable/UnregisterTickable.
    19	// Node.Tick(dtMs) iterates all registered tickables and recurses into children.
    20	// The scheduler (App) computes dtMs once per frame and passes it down — Node itself
    21	// has no concept of wall-clock time or animation.
    22	//
    23	// Lifecycle tracking (mIsNew, mInitializedSlots) lets capabilities react to
    24	// node creation, reuse, state cleanup, and disposal via the StateSlotRegistry.

   258		// HoverState removed), and its stale slot data should be cleaned so mState

   271				// and release any external resources before the slot is removed.

   285		// - Reused node (isNew=false): fires OnCreate for slots that appear for the
   286		//   first time on this node, and OnReuse for slots that already existed.
   287		//

   289		// in the reconciliation flow), and OnDispose by Dispose().
   290		public void OnReconciled(bool isNew, HashSet<int> liveSlots)

   295				if (isNew)
   296				{
   297					if (lc.OnCreate != null)
   298						lc.OnCreate(this, slot);
   299					mInitializedSlots.Add(slot);
   300				}
   301				else if (!mInitializedSlots.Contains(slot))
   302				{
   303					// Slot is live on this widget for the first time (even though
   304					// the node existed before — the widget gained a new capability).
   305					if (lc.OnCreate != null)
   306						lc.OnCreate(this, slot);
   307					mInitializedSlots.Add(slot);
   308				}
   309				else
   310				{
   311					if (lc.OnReuse != null)
   312						lc.OnReuse(this, slot);
   313				}
   314			}
   315	
   316			mIsNew = false;
   317		}

   324		// resources (timers, native handles, subscriptions). Children are NOT
   325		// recursively disposed here: the orphan loop in App.ReconcileTree disposes
   326		// every orphaned Node individually. A reused child is reattached via
   327		// ReconcileWidget, which calls Children.Clear() first to drop the old
   328		// reference. Recursing here would double-free orphaned children.
   329		public void Dispose()
   330		{
   331			// Delete tickables first (each destructor unregisters from mTickables).
   332			// Snapshot so the list isn't mutated under iteration.
   333			List<ITickable> snapshot = scope .();
   334			for (let t in mTickables)
   335				snapshot.Add(t);
   336			for (let t in snapshot)
   337				delete t;
   338			mTickables.Clear();
   339	
   340			// Fire OnDispose for every slot that still has data in this node.
   341			for (let pair in mState)
   342			{
   343				let lc = StateSlotRegistry.Get(pair.key);
   344				if (lc.OnDispose != null)
   345					lc.OnDispose(this, pair.key);
   346			}
   347			Children.Clear();
   348		}

     6	// A typed, named, discoverable key into the generic per-Node state dictionary.
     7	// Capabilities define their own static readonly StateSlot<T> fields (framework
     8	// slots 1–99 live in StateSlots; user slots should start at 100). The slot is
     9	// just a typed key + accessor pair — it owns no storage itself. The underlying
    10	// Dictionary<int, Variant> on Node stores the actual values.

    12	// Usage from a tracker or property:
    13	//   StateSlots.Hover.Set(node, true);
    14	//   bool hovered = StateSlots.Hover.Get(node);
    15	//
    16	// T : struct limits typed accessors to value types (bool, int, float, etc.).
    17	// The underlying Variant storage also supports heap objects, but heap-type
    18	// StateSlot support is deferred until a use case demands it.

    20	{

    22		public StringView Name;

    35				return default;

    51	// one of these; user capabilities define their own StateSlot<T> fields starting
    52	// at 100. Slots are preserved during pruning only when a capability on the
    53	// widget declares them via IHasStateSlots (behaviors) or IInteractionState
    54	// (properties) — no slot is "always live."
    55	//
    56	// Slot  |  Name       | Type          | Used by
    57	// ------+-------------+---------------+----------
    58	//   1   |  Hover      | bool          | HoverHandler / HoverState
    59	//   2   |  Press      | bool          | PressHandler / PressState
    60	//   3   |  Scroll     | float         | ScrollHandlerBehavior / ScrollView
    61	//   4   |  DragOffset | DragOffset    | DragHandlerBehavior / RenderDragOffset
    62	//   5   |  Focus      | bool          | FocusHandler
    63	public static class StateSlots
    64	{
    65		public static readonly StateSlot<bool> Hover = .(1, "Hover");
    66		public static readonly StateSlot<bool> Press = .(2, "Press");
    67		public static readonly StateSlot<float> Scroll = .(3, "Scroll");
    68		public static readonly StateSlot<DragOffset> DragOffset = .(4, "DragOffset");
    69		public static readonly StateSlot<bool> Focus = .(5, "Focus");

    72		{

    74			// with no external resources, so all four callbacks are null — the
    75			// registration exists to prove the lifecycle mechanism and serve as
    76			// documentation for user capabilities.
    77			StateSlotRegistry.Register(1, .() { }); // Hover

     1	using System;
     2	using System.Collections;
     3	
     4	namespace WordUI;
     5	
     6	// Retained animation-clock manager for a single Node. Created lazily on first
     7	// animated state-blend and registered as an ITickable on the node. Manages a
     8	// pool of per-StateId clocks — each clock tracks eased progress toward a
     9	// boolean target over a duration with an easing curve. Self-prunes clocks that
    10	// settle at P=0 with Target=false (fully inactive, no longer needed).
    11	//
    12	// StyleResolve.Factor() fetches this via node.GetTickable<AnimationState>()
    13	// and calls Blend() for each animated patch record. The draw loop is the only
    14	// consumer; layout never queries clocks.
    15	//
    16	// Memory: heap-allocated (new), owned by the Node's mTickables list via the
    17	// ITickable registration. Deleted by Node.Dispose() during the tickable
    18	// snapshot-delete pass. The destructor unregisters from the node.
    19	public class AnimationState : ITickable
    20	{
    21		// Per-stateId clock: eased progress P moves toward Target=1 or Target=0
    22		// at a rate of 1/DurationMs per tick. When Target=false and P reaches 0,
    23		// the clock is pruned (removed from mClocks).
    24		struct Clock
    25		{
    26			public float P;          // current eased-input progress, [0,1]
    27			public bool Target;      // true = animate toward 1, false = toward 0
    28			public int DurationMs;
    29			public Easing Curve;
    30		}
    31	
    32		Dictionary<int, Clock> mClocks = new .() ~ delete _;
    33		Node mNode;
    34	
    35		public this(Node node)
    36		{
    37			mNode = node;
    38		}
    39	
    40		public ~this()
    41		{
    42			if (mNode != null)
    43			{
    44				mNode.UnregisterTickable(this);
    45				mNode = null;
    46			}
    47		}
    48	
    49		// ---- ITickable ----
    50	
    51		public bool NeedsTick
    52		{
    53			get
    54			{
    55				for (let pair in mClocks)
    56				{
    57					let c = pair.value;
    58					if (c.Target && c.P < 1f)
    59						return true;
    60					if (!c.Target && c.P > 0f)
    61						return true;
    62				}
    63				return false;
    64			}
    65		}
    66	
    67		public void Tick(int dtMs)
    68		{
    69			if (dtMs <= 0)
    70				return;
    71	
    72			// Collect-then-apply: never mutate mClocks while iterating it.
    73			List<(int key, Clock clock)> updates = scope .();
    74			List<int> settled = scope .();
    75			for (var pair in mClocks)
    76			{
    77				var c = pair.value;
    78				let step = c.DurationMs > 0 ? (float)dtMs / c.DurationMs : 1f;
    79				if (c.Target)
    80					c.P = Math.Min(1f, c.P + step);
    81				else
    82					c.P = Math.Max(0f, c.P - step);
    83				updates.Add((pair.key, c));
    84				if (!c.Target && c.P <= 0f)
    85					settled.Add(pair.key);
    86			}
    87			for (let (key, clock) in updates)
    88				mClocks[key] = clock;
    89			for (let k in settled)
    90				mClocks.Remove(k);
    91		}
    92	
    93		// ---- Blend API ----
    94	
    95		// Ensure a clock exists for `stateId`, update its target/params, and return
    96		// the current eased blend factor in [0,1] (0 = base, 1 = fully at target).
    97		// Called by StyleResolve.Factor() at draw time for each animated patch.
    98		public float Blend(int stateId, bool active, int durationMs, Easing curve)
    99		{
   100			Clock c;
   101			if (!mClocks.TryGetValue(stateId, out c))
   102				c = .() { P = 0f, Target = active, DurationMs = durationMs, Curve = curve };
   103			c.Target = active;
   104			c.DurationMs = durationMs;
   105			c.Curve = curve;
   106			mClocks[stateId] = c;
   107			return c.Curve.Apply(c.P);
   108		}
   109	
   110		// ---- Static helpers ----
   111	
   112		// Returns the AnimationState for a node, creating and registering it
   113		// lazily on first access. Returns null when node is null.
   114		public static AnimationState Get(Node node)
   115		{
   116			if (node == null)
   117				return null;
   118			for (let t in node.Tickables)
   119			{
   120				let existing = t as AnimationState;
   121				if (existing != null)
   122					return existing;
   123			}
   124			let created = new AnimationState(node);
   125			node.RegisterTickable(created);
   126			return created;
   127		}
   128	}

     1	using System;
     2	using System.Collections;
     3	
     4	namespace WordUI;
     5	
     6	// Single ITracker replacing the four monolithic trackers (HoverTracker, PressTracker,
     7	// DragTracker, ScrollTracker). Hit-tests to find the target widget, then offers the
     8	// event to every IInputHandler behavior on that widget (and its ancestors, walking
     9	// leaf→root). The first handler returning true from HandleEvent on MouseDown claims
    10	// the pointer capture; subsequent events route exclusively to that widget's behaviors
    11	// until MouseUp releases it.
    12	//
    13	// Wheel events always route through the hit chain regardless of capture (they're
    14	// discrete impulses, not part of a press-drag gesture). Key events route to the
    15	// focused widget (App.FocusedKey).
    16	//
    17	// User-authored interaction types need only implement IInputHandler on a Behavior
    18	// and attach it to a widget — no tracker registration, no framework modification.
    19	class InputDispatcher : ITracker
    20	{
    21		int mHoveredKey = 0;   // 0 = nothing hovered (for built-in hover tracking)
    22	
    23		public bool HandleEvent(Widget root, InputEvent event)
    24		{
    25			let app = App.Instance;
    26	
    27			// ---- Key events: route to focused widget ----
    28			if (event.Type == .KeyDown)
    29			{
    30				if (app.FocusedKey != 0)
    31				{
    32					let w = FindByKey(root, app.FocusedKey);
    33					if (w != null)
    34					{
    35						for (let b in w.Behaviors)
    36						{
    37							let h = b as IInputHandler;
    38							if (h != null && h.HandleEvent(root, w, event))
    39								return true;
    40						}
    41					}
    42				}
    43				return false;
    44			}
    45	
    46			// ---- Wheel events: always hit chain, never captured ----
    47			if (event.Type == .MouseWheel)
    48			{
    49				return DispatchToChain(root, event.X, event.Y, event);
    50			}
    51	
    52			// ---- MouseDown: hit chain; first true return claims capture ----
    53			if (event.Type == .MouseDown)
    54			{
    55				let consumed = DispatchToChain(root, event.X, event.Y, event);
    56				// The handler that consumed the event is on the hit widget; the
    57				// InputDispatcher records the capture key. The handler itself doesn't
    58				// need to know about capture — it just returns true.
    59				if (consumed)
    60				{
    61					let hit = HitTest.At(root, event.X, event.Y);
    62					if (hit != null)
    63						app.CaptureBegin(hit.StableKey, event.X, event.Y);
    64				}
    65				return consumed;
    66			}
    67	
    68			// ---- MouseUp: release capture, then dispatch ----
    69			if (event.Type == .MouseUp)
    70			{
    71				if (app.CaptureActive)
    72				{
    73					let capKey = app.CaptureKey;
    74					app.CaptureEnd();
    75					let w = FindByKey(root, capKey);
    76					if (w != null)
    77					{
    78						for (let b in w.Behaviors)
    79						{
    80							let h = b as IInputHandler;
    81							if (h != null)
    82								h.HandleEvent(root, w, event);
    83						}
    84					}
    85					return false;
    86				}
    87				return DispatchToChain(root, event.X, event.Y, event);
    88			}
    89	
    90			return false;
    91		}
    92	
    93		public bool HandleMotion(Widget root, float x, float y)
    94		{
    95			let app = App.Instance;
    96	
    97			// ---- Captured: route motion exclusively to capture target ----
    98			if (app.CaptureActive)
    99			{
   100				let w = FindByKey(root, app.CaptureKey);
   101				if (w != null)
   102				{
   103					for (let b in w.Behaviors)
   104					{
   105						let h = b as IInputHandler;
   106						if (h != null && h.HandleMotion(root, w, x, y))
   107							return true;
   108					}
   109				}
   110				return true; // capture always consumes motion (suppresses hover walk)
   111			}
   112	
   113			// ---- No capture: built-in hover tracking + chain dispatch ----
   114			// Hover enter/leave is fundamental enough that the dispatcher handles it
   115			// directly, writing StateSlots.Hover on the retained Node tree. Per-widget
   116			// HoverHandler behaviors (if attached) receive motion for additional work.
   117			let hit = HitTest.At(root, x, y);
   118			let key = hit != null ? hit.StableKey : 0;
   119			if (key != mHoveredKey)
   120			{
   121				// Clear previous hover.
   122				if (mHoveredKey != 0)
   123				{
   124					let prev = app.FindNode(mHoveredKey);
   125					StateSlots.Hover.Set(prev, false);
   126					prev?.MarkNeedsPaint();
   127				}
   128				// Set new hover.
   129				if (key != 0)
   130				{
   131					let node = app.FindNode(key);
   132					StateSlots.Hover.Set(node, true);
   133					node?.MarkNeedsPaint();
   134				}
   135				mHoveredKey = key;
   136				app.RequestRepaint();
   137			}
   138	
   139			return DispatchMotionToChain(root, x, y);
   140		}
   141	
   142		// ---- Chain dispatch helpers ----
   143	
   144		// Hit-test at (x,y), then walk leaf→root offering the event to IInputHandler
   145		// behaviors on each widget. Returns true if any handler consumed the event.
   146		static bool DispatchToChain(Widget root, float x, float y, InputEvent event)
   147		{
   148			List<Widget> chain = scope .();
   149			BuildChain(root, x, y, chain);
   150			for (var w in chain)
   151			{
   152				for (let b in w.Behaviors)
   153				{
   154					let h = b as IInputHandler;
   155					if (h != null && h.HandleEvent(root, w, event))
   156						return true;
   157				}
   158			}
   159			return false;
   160		}
   161	
   162		// Same chain walk for motion events.
   163		static bool DispatchMotionToChain(Widget root, float x, float y)
   164		{
   165			List<Widget> chain = scope .();
   166			BuildChain(root, x, y, chain);
   167			for (var w in chain)
   168			{
   169				for (let b in w.Behaviors)
   170				{
   171					let h = b as IInputHandler;
   172					if (h != null && h.HandleMotion(root, w, x, y))
   173						return true;
   174				}
   175			}
   176			return false;
   177		}
   178	
   179		// Builds the ancestor chain from the leaf widget at (x,y) up to (but not
   180		// including) root: [leaf, parent, grandparent, ...]. The leaf is deepest;
   181		// iterating forward dispatches leaf-first.
   182		static void BuildChain(Widget root, float x, float y, List<Widget> chain)
   183		{
   184			FindChain(root, x, y, chain);
   185		}
   186	
   187		// Recursively walks the tree. When a widget containing (x,y) is found, it
   188		// recurses into children first (so leaves come first in the chain), then
   189		// adds itself. Returns true when the target leaf was found anywhere in the
   190		// subtree, so ancestors above can add themselves.
   191		static bool FindChain(Widget w, float x, float y, List<Widget> chain)
   192		{
   193			if (w == null)
   194				return false;
   195	
   196			bool hit = x >= w.mRect.X && x < w.mRect.Right &&
   197					   y >= w.mRect.Y && y < w.mRect.Bottom;
   198	
   199			let children = w.GetComponent<Children>();
   200			bool childHit = false;
   201			if (children != null)
   202			{
   203				for (let child in children.Items)
   204				{
   205					if (FindChain(child, x, y, chain))
   206						childHit = true;
   207				}
   208			}
   209	
   210			if (hit)
   211			{
   212				// Add leaf first (post-order), then ancestors bubble up.
   213				chain.Add(w);
   214				return true;
   215			}
   216	
   217			return childHit;
   218		}
   219	
   220		// Find a widget by StableKey anywhere in the tree.
   221		static Widget FindByKey(Widget w, int key)
   222		{
   223			if (w == null)
   224				return null;
   225			if (w.StableKey == key)
   226				return w;
   227			let children = w.GetComponent<Children>();
   228			if (children != null)
   229			{
   230				for (let child in children.Items)
   231				{
   232					let found = FindByKey(child, key);
   233					if (found != null)
   234						return found;
   235				}
   236			}
   237			return null;
   238		}
   239	}

    23		{
    24			let line = scope String();
    25			line.Append(' ', depth * 2);
    26			line.Append(scope $"<{w.Name}");
    27	
    28			let props = scope String();
    29	
    30			// Components (pipeline participants)
    31			for (let c in w.Components)
    32			{
    33				let d = scope String();
    34				c.Describe(d);
    35	
    36				if (!d.IsEmpty)
    37				{
    38					if (!props.IsEmpty) props.Append(' ');
    39					props.Append(d);
    40				}
    41			}
    42	
    43			// Properties (pure data)
    44			for (let p in w.Properties)
    45			{
    46				let d = scope String();
    47				p.Describe(d);
    48	
    49				if (!d.IsEmpty)
    50				{
    51					if (!props.IsEmpty) props.Append(' ');
    52					props.Append(d);
    53				}
    54			}
    55	
    56			// Behaviors (event handlers)
    57			for (let b in w.Behaviors)
    58			{
    59				let d = scope String();
    60				b.Describe(d);
    61	
    62				if (!d.IsEmpty)
    63				{
    64					if (!props.IsEmpty) props.Append(' ');
    65					props.Append(d);
    66				}
    67			}
    68	
    69			if (!props.IsEmpty) props.Append(' ');
    70			props.AppendF(" Padding={}", w.Padding);
    71	
    72			if (!props.IsEmpty) props.Append(' ');
    73			props.AppendF(scope $" Key={w.Key}");
    74	
    75			if (!props.IsEmpty) props.Append(' ');
    76			props.Append(scope $" Rect=({w.mRect.X},{w.mRect.Y}, {w.mRect.Width}x{w.mRect.Height})");
    77

     6	// Wheel/scroll interaction handler. Carries its own ScrollHandler delegate — no
     7	// separate delegate-holder behavior. On MouseWheel, finds the deepest ScrollView-bearing
     8	// widget under the pointer, adjusts its persistent scroll offset in the retained Node
     9	// (clamped to content extent), fires the handler, and requests a repaint — no rebuild.
    10	// The offset is applied at draw by ScrollView.GetRenderOffset.
    11	//
    12	// Does NOT claim capture (returns false from HandleEvent). Wheel events are discrete
    13	// impulses, not a press-drag gesture, so they always route through the hit chain
    14	// regardless of any active capture.
    15	//
    16	// Implements IHasStateSlots so the reconciliation pass preserves the Scroll state
    17	// slot on this widget's Node.
    18	//
    19	// State slot: StateSlots.Scroll (3) — float stored on the retained Node.
    20	public class ScrollHandlerBehavior : Behavior, IInputHandler, IHasStateSlots
    21	{
    22		public ScrollHandler Handler;   // null = no scroll callback
    23		const float StepPx = 30f;       // pixels of scroll per wheel notch
    24	
    25		public bool HandleEvent(Widget root, Widget owner, InputEvent event)
    26		{
    27			if (event.Type != .MouseWheel)
    28				return false;
    29	
    30			let hit = FindScrollableAt(root, event.X, event.Y);
    31			if (hit == null)
    32				return false;
    33	
    34			let stack = hit.GetComponent<StackLayout>();
    35			bool vertical = stack == null || stack.Axis == .Vertical;
    36	
    37			// Viewport = the container's own inner box (its allocated rect minus
    38			// padding). Content extent = how far the children reach past the inner
    39			// origin (they are laid out at natural, unclamped positions and overflow).
    40			float pad = hit.Padding;
    41			float viewportMain = (vertical ? hit.mRect.Height : hit.mRect.Width) - 2 * pad;
    42			float innerOrigin = (vertical ? hit.mRect.Y : hit.mRect.X) + pad;
    43	
    44			float contentEnd = innerOrigin;
    45			let children = hit.GetComponent<Children>();
    46			if (children != null)
    47			{
    48				for (let child in children.Items)
    49				{
    50					let edge = vertical ? child.mRect.Bottom : child.mRect.Right;
    51					contentEnd = Math.Max(contentEnd, edge);
    52				}
    53			}
    54			float maxOffset = Math.Max(0, (contentEnd - innerOrigin) - viewportMain);
    55	
    56			let app = App.Instance;
    57			let node = app.FindNode(hit.StableKey);
    58			let current = node != null ? StateSlots.Scroll.Get(node) : 0f;
    59	
    60			// SDL wheel y is positive away from the user; scrolling away reveals
    61			// earlier content, i.e. a smaller offset.
    62			float next = Math.Clamp(current - event.WheelY * StepPx, 0, maxOffset);
    63			if (node != null)
    64				StateSlots.Scroll.Set(node, next);
    65			node?.MarkNeedsPaint();
    66	
    67			// Fire the scroll handler on the scrollable widget (hit), not owner.
    68			let sh = hit.GetBehavior<ScrollHandlerBehavior>();
    69			if (sh != null && sh.Handler != null)
    70				sh.Handler(next);
    71	
    72			app.RequestRepaint();
    73			return false;
    74		}
    75	
    76		public bool HandleMotion(Widget root, Widget owner, float x, float y)
    77		{
    78			return false;
    79		}
    80	
    81		public void CollectSlots(HashSet<int> slots)
    82		{
    83			slots.Add(StateSlots.Scroll.Id);
    84		}

    86		public override void Describe(String buf)
    87		{
    88			buf.Append(" OnScroll");
    89		}

    91		// Deepest widget carrying a ScrollView whose rect contains the point.
    92		static Widget FindScrollableAt(Widget root, float x, float y)
    93		{
    94			return Find(root, x, y, null);
    95		}
    96	
    97		static Widget Find(Widget w, float x, float y, Widget best)
    98		{
    99			if (w == null)
   100				return best;
   101	
   102			Widget current = best;
   103			if (x >= w.mRect.X && x < w.mRect.Right &&
   104				y >= w.mRect.Y && y < w.mRect.Bottom &&
   105				w.GetComponent<ScrollView>() != null)
   106			{
   107				current = w;
   108			}
   109	
   110			let children = w.GetComponent<Children>();
   111			if (children != null)
   112			{
   113				for (let child in children.Items)
   114					current = Find(child, x, y, current);
   115			}
   116			return current;
   117		}

     5	// Fired when a key is pressed while this widget holds focus.
     6	// keycode: SDL keycode (platform-independent virtual key)
     7	// modifiers: bitmask of SDL keymods (Shift, Ctrl, Alt, etc.)
     8	public delegate void KeyHandler(int keycode, int modifiers);
     9	
    10	// Keyboard input handler. Carries its own KeyHandler delegate — no separate
    11	// delegate-holder behavior. When a KeyDown event arrives and this widget holds
    12	// focus (checked via App.FocusedKey == owner.StableKey), fires the handler.
    13	// Does NOT declare state slots — focus state is managed by FocusHandler.
    14	public class KeyboardHandler : Behavior, IInputHandler
    15	{
    16		public KeyHandler Handler;   // null = no key callback
    17	
    18		public bool HandleEvent(Widget root, Widget owner, InputEvent event)
    19		{
    20			if (event.Type != .KeyDown)
    21				return false;
    22	
    23			if (App.Instance.FocusedKey != owner.StableKey)
    24				return false;
    25	
    26			if (Handler != null)
    27				Handler(event.KeyCode, event.KeyModifiers);
    28			return false;
    29		}
    30	
    31		public bool HandleMotion(Widget root, Widget owner, float x, float y)
    32		{
    33			return false;
    34		}
    35	
    36		public override void Describe(String buf)
    37		{
    38			buf.Append(" OnKey");
    39		}
    40	}

    42	extension UI
    43	{
    44		// Makes the stack-top widget receive keyboard events when focused.
    45		// Attaches KeyboardHandler (which owns the KeyHandler delegate directly —
    46		// one behavior, no separate holder). Pair with Focusable() to make the
    47		// widget focusable.
    48		public static void OnKey(KeyHandler h)
    49		{
    50			let w = Top("OnKey");
    51			var kh = w.GetBehavior<KeyboardHandler>();
    52			if (kh == null)
    53			{
    54				kh = new:BuildAlloc KeyboardHandler();
    55				w.AttachBehavior(kh);
    56			}
    57			RegisterHandler(h);
    58			kh.Handler = h;
    59		}
    60	
    61		// Makes the stack-top widget focusable: clicking it gives it keyboard focus.
    62		// Attaches FocusHandler. Pair with OnKey() to also receive key events.
    63		public static void Focusable()

    65			let w = Top("Focusable");
    66			var fh = w.GetBehavior<FocusHandler>();

    38	// M4 demo: a functional todo app — add an item, remove an item, and scroll the
    39	// list. The list lives in a Scrollable container with a bounded height and Clip,
    40	// so it overflows into a scroll viewport whose offset survives the rebuild each
    41	// add/remove triggers. No text input yet (M5), so item text is auto-generated.
    42	class TodoSDLApp : SDLApp
    43	{
    44		// App-owned state: id keeps a stable identity per item as the list churns.
    45		List<(int id, String text)> mTodos = new .() ~ {
    46			for (let t in _) delete t.text;
    47			delete _;
    48		};
    49		int mNextId = 1;
    50	
    51		public this()
    52		{
    53			ClearColor = .(20, 40, 60, 255);
    54			for (int i = 0; i < 8; i++)
    55				AddTodo();
    56		}
    57	
    58		void AddTodo()
    59		{
    60			let n = mNextId++;
    61			mTodos.Add((n, new String(scope $"Task {n}")));
    62			Invalidate();
    63		}
    64	
    65		void RemoveTodo(int id)
    66		{
    67			for (int i = 0; i < mTodos.Count; i++)
    68			{
    69				if (mTodos[i].id == id)
    70				{
    71					delete mTodos[i].text;
    72					mTodos.RemoveAt(i);
    73					break;
    74				}
    75			}
    76			Invalidate();
    77		}
    78	
    79		protected override void Build()
    80		{
    81			VStack(scope () => {
    82				TitleLabel("Todos"); Center();
    83	
    84				HStack(scope () => {
    85					Button("Add"); {
    86						TextColor(.White);
    87						IfPressed(scope () => { Fill(.(120, 120, 120, 255)); });
    88						OnClick(new () => { AddTodo(); });
    89					}
    90					Spacer();
    91					Label(scope $"{mTodos.Count} items"); AlignEnd(); TextColor(.(180, 180, 180, 255));
    92				}); Spacing(8);
    93	
    94				// The scroll viewport: a vertical Scrollable, clipped to a fixed
    95				// height. Its children overflow and the wheel scrolls them.
    96				Scrollable(scope () => {
    97					for (let t in mTodos)
    98					{
    99						let id = t.id;   // per-iteration copy for the delete closure
   100						TodoRow(id, t.text, new () => { RemoveTodo(id); });
   101					}
   102				}); Height(.Fixed(240)); Spacing(6); Clip();
   103			}); Padding(16);
   104		}
   105	}
